cmake_minimum_required(VERSION 3.14)
project(deskup_configuration_tests LANGUAGES CXX)
add_executable(configuration_test runtime_configuration_test.cpp ../../windows/runtime_configuration.cpp)
target_compile_features(configuration_test PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(configuration_test PRIVATE /W4 /WX /wd4100)
endif()
target_include_directories(configuration_test PRIVATE ../../windows "${FLUTTER_WRAPPER_INCLUDE}")
enable_testing()
add_test(NAME configuration COMMAND configuration_test)
